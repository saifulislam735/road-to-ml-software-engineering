import asyncio

async def call_llm(prompt: str) -> str:
    # async means: while waiting for LLM response, do other things
    await asyncio.sleep(0.1)  # simulating network wait
    return f"Response to: {prompt}"

async def process_batch(prompts: List[str]) -> List[str]:
    # KEY: Run multiple LLM calls in PARALLEL — 5x faster than sequential!
    results = await asyncio.gather(*[call_llm(p) for p in prompts])
    return results

# Sequential: 10 LLM calls × 2 seconds each = 20 seconds total
# Parallel:   10 LLM calls with asyncio.gather = ~2 seconds total!
asyncio.run(process_batch(["Q1", "Q2", "Q3"]))

# ── 4. Retry with Exponential Backoff — LLM APIs fail sometimes ──
import time, random

def call_with_retry(func, max_retries: int = 3):
    for attempt in range(max_retries):
        try:
            return func()
        except Exception as e:
            if attempt == max_retries - 1:
                raise  # last attempt — raise the error
            wait = (2 ** attempt) + random.random()  # 1s, 2s, 4s + jitter
            print(f"Attempt {attempt+1} failed: {e}. Retrying in {wait:.1f}s")
            time.sleep(wait)

Chapter 04 — Core Concepts

🧠 LLM Fundamentals — The Building Blocks

Before using LangChain or RAG, understand how LLMs work internally. These concepts appear in every interview.

Concept 1: Tokens — How LLMs Read Text

LLMs don't read words or characters. They read tokens — chunks that can be parts of words, whole words, or punctuation. "tokenization" becomes ["token", "iz", "ation"] — 3 tokens.

📏
Token Rule of Thumb

1 token ≈ 4 characters ≈ 0.75 words. "Hello world" = 2 tokens. A full page = ~500 tokens. GPT-4o context = 128,000 tokens = ~100,000 words. API pricing is per 1 million tokens.

Pythontokens.py — Count tokens and estimate cost
import tiktoken
from typing import List

def count_tokens(text: str, model: str = "gpt-4o") -> int:
    """Count tokens to estimate API cost BEFORE calling LLM"""
    enc = tiktoken.encoding_for_model(model)
    return len(enc.encode(text))

def estimate_cost(text: str, output_tokens: int = 1000) -> dict:
    """Estimate API cost for a prompt + expected output"""
    input_tokens = count_tokens(text)
    # GPT-4o pricing (2026): $5/1M input, $15/1M output
    input_cost  = (input_tokens  / 1_000_000) * 5.0
    output_cost = (output_tokens / 1_000_000) * 15.0
    return {
        "input_tokens": input_tokens,
        "total_cost_usd": round(input_cost + output_cost, 6)
    }

text = "Explain transformers in machine learning."
print(count_tokens(text))   # 7
print(estimate_cost(text))  # ~$0.000045

# Real-world impact: 5000-token RAG context x 10,000 daily queries
# = 50M tokens/day = $250/day = $7,500/month.
# Always measure BEFORE scaling!

Concept 2: Embeddings — Text as Meaning-Vectors

An embedding converts text into a list of numbers (vector) representing its meaning. Similar meanings produce similar vectors. This is the foundation of all RAG and semantic search.

Pythonembeddings.py — Create and compare text embeddings
from openai import OpenAI
import numpy as np

client = OpenAI()

def embed(text: str) -> List[float]:
    """Convert text to 1536-dimensional float vector"""
    res = client.embeddings.create(
        model="text-embedding-3-small",
        input=text
    )
    return res.data[0].embedding

def cosine_similarity(a, b) -> float:
    """Similarity score: 0.0 = totally different, 1.0 = identical meaning"""
    va, vb = np.array(a), np.array(b)
    return float(np.dot(va, vb) / (np.linalg.norm(va) * np.linalg.norm(vb)))

e1 = embed("How do I train a neural network?")
e2 = embed("Steps to build a deep learning model")
e3 = embed("What is the recipe for biryani?")

print(cosine_similarity(e1, e2))  # 0.93 — very similar despite different words!
print(cosine_similarity(e1, e3))  # 0.41 — very different topics

# This is WHY RAG works: "vacation days" and "annual leave entitlement"
# have 0.91 similarity even though they share zero words.

Chapter 05 — LangChain

⛓️ LangChain & LCEL — The AI Engineering Backbone

LangChain is in almost every production AI app. Master it step by step, from simple chains to complex agents.

Step 1 — The LCEL Pipe Operator

LCEL uses the | pipe operator to chain components. Data flows left to right — each component receives the previous one's output.

Pythonlcel.py — Chain building, streaming, and batching
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

llm   = ChatOpenAI(model="gpt-4o-mini", temperature=0.7)
prompt = ChatPromptTemplate.from_messages([
    ("system", "You are a helpful {role}."),
    ("human",  "{question}")
])

# Build chain: prompt -> LLM -> parse output to string
chain = prompt | llm | StrOutputParser()

# Invoke: single synchronous call
r = chain.invoke({"role": "Python expert", "question": "What is a decorator?"})

# Stream: print tokens as they arrive (better UX)
for chunk in chain.stream({"role": "tutor", "question": "Explain async/await"}):
    print(chunk, end="", flush=True)

# Batch: multiple inputs in parallel (much faster than sequential!)
inputs = [
    {"role": "tutor", "question": "What is Python?"},
    {"role": "tutor", "question": "What is JavaScript?"},
]
responses = chain.batch(inputs)  # runs all in parallel!

# Async: for FastAPI and other async web frameworks
import asyncio
async def main():
    r = await chain.ainvoke({"role": "expert", "question": "Hi"})
    return r

Pythonagent.py — Agent with custom tools
from langchain_core.tools import tool
from langchain_openai import ChatOpenAI
from langchain.agents import AgentExecutor, create_tool_calling_agent
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder

@tool
def calculate(expression: str) -> str:
    """Evaluate math. Examples: '15% of 8500', '2**10', 'sqrt(144)'"""
    import math
    try: return str(eval(expression, {"__builtins__": {}}, vars(math)))
    except Exception as e: return f"Error: {e}"

@tool
def get_weather(city: str) -> str:
    """Get current weather for a city."""
    return f"{city}: 28°C, Sunny"  # call real API in production

tools = [calculate, get_weather]
llm   = ChatOpenAI(model="gpt-4o", temperature=0)

prompt = ChatPromptTemplate.from_messages([
    ("system", "You are a helpful assistant. Use tools for math and weather. Never guess."),
    MessagesPlaceholder("chat_history", optional=True),
    ("human", "{input}"),
    MessagesPlaceholder("agent_scratchpad")
])

agent    = create_tool_calling_agent(llm, tools, prompt)
executor = AgentExecutor(agent=agent, tools=tools, verbose=True, max_iterations=5)

result = executor.invoke({"input": "What is 15% of 8500 and the weather in Dhaka?"})
print(result["output"])
# Agent: 1) calls calculate("8500*0.15") -> "1275.0"
#        2) calls get_weather("Dhaka") -> "Dhaka: 28°C, Sunny"
#        3) combines into natural response
Chapter 06 — RAG

🔍 RAG Systems — Retrieval-Augmented Generation

RAG lets your LLM answer from YOUR documents. It is the number-one most commonly built AI system in industry. Master this completely — it appears in nearly every AI engineering interview.

The RAG Pipeline — Every Step Explained

INDEXING PHASE (runs once when documents change) 📄 Your documents (PDF, Word, Web, Database) | v [Text Extraction] -- PyPDF, Unstructured, BeautifulSoup | v [Chunking] -- Split into 500-1000 char pieces with 200-char overlap | WHY: LLMs have context limits. Smaller = more precise retrieval. v [Embedding] -- Convert each chunk to a vector (list of numbers) | text-embedding-3-small: 1536 numbers per chunk v [Vector DB Storage] -- ChromaDB / Pinecone / Qdrant Stores: (chunk_text, embedding_vector, metadata) RETRIEVAL PHASE (runs for every user query) 💬 User Question: "What is the vacation policy?" | v [Embed Question] -- Same model as indexing! Critical requirement. | v [Vector Search] -- Find top-K chunks most similar to question | Cosine similarity: which chunks are closest? v [Reranking] -- (Optional but recommended) Re-score with better model | Cohere Rerank, BGE Reranker v [LLM + Context] -- "Answer this using ONLY these chunks: {context}..." | v Final Answer (grounded in your documents, not hallucinated)

Step-by-Step: Build Your First RAG System

Pythonrag_basic.py — Complete working RAG system
from langchain_community.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import OpenAIEmbeddings, ChatOpenAI
from langchain_community.vectorstores import Chroma
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.runnables import RunnablePassthrough
from langchain_core.output_parsers import StrOutputParser

# STEP 1: Load document
loader = PyPDFLoader("company_handbook.pdf")
pages  = loader.load()
print(f"Loaded {len(pages)} pages")

# STEP 2: Split into chunks
splitter = RecursiveCharacterTextSplitter(
    chunk_size=1000,    # ~250 tokens per chunk
    chunk_overlap=200,  # 200 chars overlap to avoid losing context at edges
    separators=["\n\n", "\n", ". ", " ", ""]  # tries splits in order
)
chunks = splitter.split_documents(pages)
print(f"Created {len(chunks)} chunks")

# STEP 3: Embed and store in vector DB
embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
vectorstore = Chroma.from_documents(
    documents=chunks,
    embedding=embeddings,
    persist_directory="./chroma_db"  # saves to disk so you don't re-embed
)

# STEP 4: Create retriever (fetches top-K relevant chunks)
retriever = vectorstore.as_retriever(
    search_type="similarity",
    search_kwargs={"k": 4}  # return 4 most similar chunks
)

# STEP 5: RAG prompt — CRITICAL to tell LLM to stay in context
rag_prompt = ChatPromptTemplate.from_template("""
You are a helpful assistant. Answer ONLY from the provided context.
If the answer is not in the context, say "I cannot find this in the documents."
Do not use your training knowledge — only use what is in the context below.

CONTEXT:
{context}

QUESTION: {question}

ANSWER:""")

def format_docs(docs):
    """Join retrieved chunks with separators so LLM sees them clearly"""
    return "\n\n---\n\n".join(
        f"Source: {d.metadata.get('source','?')} Page: {d.metadata.get('page','?')}\n{d.page_content}"
        for d in docs
    )

# STEP 6: Build RAG chain
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

rag_chain = (
    {"context": retriever | format_docs, "question": RunnablePassthrough()}
    | rag_prompt
    | llm
    | StrOutputParser()
)

# STEP 7: Query
answer = rag_chain.invoke("What is the company vacation policy?")
print(answer)

🕸️ LangGraph — Stateful Multi-Agent Systems

LangGraph lets you build agents with loops, branches, persistence, parallel execution, and human-in-the-loop. The most important advanced tool in 2026.

The Mental Model: State Machines as Flowcharts

LangGraph = flowchart for AI. Nodes are functions (what to do). Edges are connections (where to go). State is a shared dictionary that persists through every step.

LangGraph Core Concepts State (TypedDict) -- shared dict, flows through every node | [Node 1: process_input] -- reads state, does work, returns updated state | [Node 2: call_llm] -- can call LLM, tools, external APIs | [Conditional Edge] -- branch based on state value / \ [use_tools] [END] -- route to tool node or finish After tool node: edge goes BACK to call_llm -- creates the loop!
Pythonlanggraph_agent.py — Complete ReAct agent with state persistence
from langgraph.graph import StateGraph, END
from langgraph.prebuilt import ToolNode, tools_condition
from langchain_core.messages import HumanMessage
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from typing import TypedDict, Annotated, List
import operator

# STEP 1: Define State — the shared dictionary
class AgentState(TypedDict):
    # Annotated with operator.add means messages ACCUMULATE (not replaced)
    messages: Annotated[List, operator.add]
    step_count: int  # track steps to prevent infinite loops

# STEP 2: Define tools
@tool
def search_web(query: str) -> str:
    """Search the web for current information."""
    return f"Search results for '{query}': [simulated recent news]"

@tool
def run_code(code: str) -> str:
    """Execute Python code and return the result."""
    import io, contextlib
    output = io.StringIO()
    try:
        with contextlib.redirect_stdout(output):
            exec(code, {"__builtins__": {"print": print, "range": range}})
        return output.getvalue() or "Code ran with no output"
    except Exception as e:
        return f"Error: {e}"

tools = [search_web, run_code]

# STEP 3: LLM with tools bound
llm = ChatOpenAI(model="gpt-4o")
llm_with_tools = llm.bind_tools(tools)

# STEP 4: Node functions
def agent_node(state: AgentState):
    """Main agent: LLM decides next action"""
    response = llm_with_tools.invoke(state["messages"])
    return {"messages": [response], "step_count": state.get("step_count", 0) + 1}

tool_node = ToolNode(tools=tools)  # handles tool execution automatically

# STEP 5: Build graph
graph = StateGraph(AgentState)
graph.add_node("agent", agent_node)
graph.add_node("tools", tool_node)
graph.set_entry_point("agent")

# Conditional edge: if agent wants tools -> tools node; else -> END
graph.add_conditional_edges(
    "agent",
    tools_condition,  # built-in: checks if response has tool_calls
    {"tools": "tools", "__end__": END}
)
graph.add_edge("tools", "agent")  # after tool execution, back to agent

app = graph.compile()

# STEP 6: Run
result = app.invoke({
    "messages": [HumanMessage(content="Search for latest AI news and summarize it")],
    "step_count": 0
})
print(result["messages"][-1].content)

def get_employee(employee_id: int) -> dict:
    """Get employee details by ID. Returns name, role, department, email."""
    conn = sqlite3.connect("company.db")
    row = conn.execute("SELECT id,name,role,dept,email FROM employees WHERE id=?",
                       [employee_id]).fetchone()
    conn.close()
    if not row: return {"error": "Employee not found"}
    return {"id": row[0], "name": row[1], "role": row[2], "dept": row[3], "email": row[4]}

@mcp.tool()
def create_task(title: str, assignee_id: int, due_date: str, priority: str = "medium") -> str:
    """Create a task and assign it to an employee. Priority: low/medium/high."""
    conn = sqlite3.connect("company.db")
    cursor = conn.execute(
        "INSERT INTO tasks (title, assignee_id, due_date, priority) VALUES (?,?,?,?)",
        [title, assignee_id, due_date, priority]
    )
    conn.commit()
    task_id = cursor.lastrowid
    conn.close()
    return f"Task #{task_id} '{title}' created, assigned to employee {assignee_id}"

def query_tasks(status: str = "open", dept: str = "") -> list:

LLM Fundamentals (Q01–Q10)

Q01What is a token? Why does tokenization matter in practice?+

A token is a subword unit — LLMs split text into chunks that can be parts of words, whole words, or punctuation. "unhappy" → ["un","happy"] (2 tokens). "tokenization" → ["token","iz","ation"] (3 tokens). On average 1 token ≈ 0.75 words ≈ 4 characters.

Why it matters: (1) API pricing is per token — count before calling. (2) Context window limits are in tokens — 128k for GPT-4o. (3) Non-English languages tokenize less efficiently — Bengali can use 2-4x more tokens per word than English, increasing costs. (4) Long documents must be chunked because they exceed context limits. Use tiktoken to count tokens before making expensive API calls.

Q02What is temperature and when do you set it to 0 vs 0.7 vs 1.0+?+

Temperature controls the randomness of token selection. At 0, the model always picks the highest-probability next token (deterministic — same input always gives same output). At 1.0, it samples proportionally to probabilities. Above 1.0, it becomes increasingly random and incoherent.

temperature=0: data extraction, SQL generation, classification, factual Q&A, code generation. You need consistency and correctness.

temperature=0.3–0.7: summarization, translation, professional emails. Some variation acceptable but not too wild.

temperature=0.7–1.0: creative writing, brainstorming, marketing copy, story generation. Want diverse, fresh outputs each time.

Q03Explain embeddings and cosine similarity. How do they power RAG?+

An embedding is a dense vector (e.g., 1536 float numbers) that captures the semantic meaning of text. The key property: texts with similar meanings produce similar vectors in vector space.

Cosine similarity measures the angle between two vectors, returning a score from -1 to 1 (in practice 0 to 1 for text). Score of 0.95 means nearly identical meaning. Score of 0.3 means unrelated.

In RAG: At index time, each chunk is embedded and stored. At query time, the user's question is embedded with the SAME model (critical — mismatched models break everything). Then you find chunks with highest cosine similarity to the question. These are the most relevant context chunks. The magic: "vacation days" and "annual leave entitlement" produce 0.91 similarity even though they share zero words — that's semantic search.

Q04What is hallucination? How do you detect and prevent it in production?+

Hallucination is when an LLM generates factually wrong information with high confidence. It happens because LLMs optimize for statistically likely next tokens, not factual accuracy. They pattern-match instead of "knowing facts."

Prevention:

1. RAG with strong prompt: "Answer ONLY from the provided context. If not in context, say you don't know." Dramatically reduces hallucination by grounding in real documents.

2. temperature=0: More deterministic, less creative = fewer invented facts.

3. Structured output + validation: Use Pydantic schemas. If output doesn't match expected format, it likely hallucinated.

4. RAGAS faithfulness metric: Automatically extracts claims from the answer and verifies each one against the retrieved context. Low score = hallucination detected.

5. Self-consistency: Ask the same question 3x with temperature=0.5. If answers differ significantly, flag for human review.

Q05What is the difference between few-shot and zero-shot prompting?+

Zero-shot: Give the LLM the task description with no examples. Relies entirely on the model's training. Example: "Classify this review as positive/negative: 'Great product!'"

Few-shot: Provide 3-10 example input/output pairs before the actual task. The model learns the pattern from examples. Example: Show 3 (review, label) pairs, then ask to classify a new review.

When to use few-shot: When zero-shot gives wrong format, wrong style, or inconsistent results. Examples dramatically improve classification tasks, data extraction, and any task with a specific output format. Cost: each example adds tokens, so balance quality vs cost. Typically 3-5 examples is the sweet spot — more rarely helps but adds cost.

RAG Systems (Q06–Q15)

Q06Walk through the full RAG pipeline. Where are the failure points?+

Pipeline: Document loading → chunking → embedding → vector storage → query embedding → retrieval → (reranking) → LLM generation → output.

Failure point 1 — Bad chunking: Chunks too large dilute relevance; too small lose context. Fix: RecursiveCharacterTextSplitter with 1000-char chunks, 200-char overlap.

Failure point 2 — Mismatched embedding models: Index with model A, query with model B = nonsense results. Fix: always use the same embedding model for both phases. Store model name with index metadata.

Failure point 3 — Retrieval misses: User asks "refund" but doc says "return policy." Vector search may rank this lower. Fix: hybrid search (BM25 + vector).

Failure point 4 — Context window overflow: Retrieving 10 chunks × 1000 chars = 10,000 chars of context, which may exceed limits. Fix: rerank and keep top 3 only.

Failure point 5 — LLM ignores context: LLM uses training knowledge instead of retrieved context. Fix: strong system prompt "ONLY use the provided context." Measure with RAGAS faithfulness.

Q07What is hybrid search and why is it better than pure vector search?+

Hybrid search combines two retrieval methods and merges their results:

Vector search: Finds semantically similar documents based on meaning. Excellent at synonyms and paraphrases. "Neural network training" matches "deep learning model building" even with no shared words.

BM25 (keyword search): Traditional TF-IDF-based ranking. Excellent at exact matches for specific terms, product codes, names, acronyms. "ISO 9001" or "gpt-4o-mini" finds exact matches reliably.

Why hybrid wins: Vector fails on specific product names and acronyms. BM25 fails on synonyms and paraphrases. Together they cover each other's weaknesses. Real-world benchmark improvement: ~10-20% better recall vs either method alone. Implement with LangChain EnsembleRetriever: weights=[0.5, 0.5]. In production, tune weights based on your query distribution — more keyword-heavy domains benefit from higher BM25 weight.

Q08How do you evaluate a RAG system? What metrics matter?+

Use RAGAS framework. Four core metrics:

1. Faithfulness (0–1): Does the answer contain only facts from the retrieved context? Low = hallucination. Measures: extract claims from answer, verify each claim against context. Most important metric for trust.

2. Answer Relevancy (0–1): Does the answer actually address the question asked? Low = off-topic response.

3. Context Precision (0–1): Of the retrieved chunks, how many were actually relevant? Low = retrieval returning noise.

4. Context Recall (0–1): Did retrieval find all the chunks needed to answer? Low = missing key information.

Evaluation process: Build a test dataset of 50-100 (question, ideal_answer, relevant_chunks) triplets — your "golden set." Run RAGAS on each change to chunking, retrieval, or LLM. Track metrics in LangSmith across versions. Catch regressions before they reach users.

Q09What is reranking and when should you use it?+

Reranking is a two-stage retrieval process. First, retrieve the top-K candidates (fast but approximate — 8-20 chunks). Then, re-score those candidates with a more powerful but slower cross-encoder model, and keep only the top-N (3-5 chunks).

Why it works: Initial retrieval (bi-encoder) computes query and document embeddings separately. Cross-encoders (rerankers) process query+document together, capturing fine-grained interactions. Much more accurate, but too slow to run on millions of documents. The two-stage approach gets the best of both worlds.

Use Cohere Rerank: CohereRerank(model="rerank-english-v3.0", top_n=3). Works with LangChain's ContextualCompressionRetriever.

When to use: When retrieval precision is low (too many irrelevant chunks). When context window is limited (want fewest, highest-quality chunks). For high-stakes applications like legal, medical, or financial Q&A. Adds ~100-300ms latency and small API cost.

Q10How would you handle multi-document RAG where docs have contradictory information?+

Contradictory information is a real challenge in enterprise RAG. Multiple strategies:

1. Metadata-based prioritization: Store source reliability, document date, and document type in metadata. At query time, filter/boost recent or authoritative sources. Example: "policy_doc" > "meeting_notes".

2. Source attribution in prompt: Include source name and date with each chunk. Tell LLM: "If sources contradict, prefer the most recent official policy document and flag the contradiction to the user."

3. Explicit contradiction detection: After retrieval, use a cheap LLM call to detect if chunks contradict. If yes, include ALL contradicting chunks and ask the generation LLM to acknowledge and reconcile.

4. Version control for documents: When re-indexing, delete old version chunks before adding new ones. Use document_id + version as unique ID.

Vector Databases (Q11–Q16)

Q11What is HNSW? Why do vector databases use approximate nearest neighbor search?+

The problem: Finding the exact nearest neighbor in 1536-dimensional space requires computing distance to every stored vector — O(n) time. With 1 million vectors, that's 1M calculations per query, taking seconds. Too slow for real-time use.

HNSW (Hierarchical Navigable Small World): An approximate nearest neighbor (ANN) algorithm that builds a multi-layer graph. Top layers are sparse "highway" connections linking distant nodes. Bottom layers are dense "local" connections to nearby nodes. Search starts at top, quickly narrows the search space through layers, arriving at approximate nearest neighbors in O(log n) time.

Tradeoff: HNSW might miss the true closest vector 1-5% of the time. But it's 100-1000x faster than exact search. For RAG, approximate is absolutely fine — you're not doing precision science, you're finding relevant context. Every major vector DB (Pinecone, Qdrant, ChromaDB, Weaviate) uses HNSW. Key parameters to tune: ef_construction (build quality vs speed), M (graph connections per node), ef_search (query quality vs speed).

Q12Pinecone vs Qdrant vs ChromaDB vs pgvector — how do you choose?+

ChromaDB: Zero setup, in-process (no server). Perfect for learning, prototypes, demos. Runs locally with Python. Not horizontally scalable — max a few million vectors on one machine. Never use in production with multiple servers.

Pinecone: Fully managed SaaS, zero ops. Handles billions of vectors, auto-scales. Best when you want to never think about vector DB infrastructure and can afford $70+/month. Proprietary — no self-hosting.

Qdrant: Open source. Self-host with Docker or use Qdrant Cloud. Performance comparable to Pinecone. Excellent payload filtering. Free to self-host. Best for teams with DevOps capacity who want control and cost savings at scale.

pgvector: PostgreSQL extension. If you already use Postgres, just CREATE EXTENSION vector. Gives vector search inside your existing database. Simplest architecture — no extra service. Not as fast as dedicated vector DBs for billions of vectors, but perfect for millions.

Decision: Learning → ChromaDB. Production (managed) → Qdrant Cloud or Pinecone. Production (existing Postgres, <10M vectors) → pgvector.

LangGraph & Agents (Q17–Q22)

Q17Explain the ReAct pattern. How does LangGraph implement it?+

ReAct = Reason + Act. The agent alternates between (1) thinking/reasoning about what to do, (2) acting by calling a tool, (3) observing the tool result, (4) thinking again — until it has enough information to give a final answer.

Classic text-based ReAct:
Thought: I need to find Dhaka's population
Action: search_web("Dhaka population 2024")
Observation: Dhaka has 23 million people
Thought: Now I have the answer
Final Answer: Dhaka's population is approximately 23 million

LangGraph implementation: Two nodes — "agent" (calls LLM) and "tools" (executes tool calls). A conditional edge: if response has tool_calls → go to tools node, then back to agent. If response has no tool_calls → END. This creates the Reason-Act loop without any text parsing. The LLM generates structured tool_calls objects natively via tool binding.

Q18How do you prevent infinite loops in LangGraph agents?+

Infinite loops happen when an agent calls tools repeatedly without converging on an answer.

1. recursion_limit: LangGraph's built-in safety: app.invoke(input, config={"recursion_limit": 10}). Default is 25 steps. When exceeded, LangGraph raises GraphRecursionError.

2. Step counter in state: Add step_count: int = 0 to your state TypedDict. Increment in every node. Add conditional edge: if step_count > max_steps → END with error message.

3. Detect repeated tool calls: Track last tool call in state. If agent calls the same tool with the same arguments twice, stop and return what you have.

4. Human-in-the-loop: For high-stakes actions (send email, delete data), use interrupt_before=["tools"] checkpoint. Agent pauses, human reviews and approves, then resumes.

5. Tool result quality check: After each tool call, have the LLM rate whether it got useful information (0-1). If rating stays low for 3 consecutive calls, stop.

Q19What is the difference between LangChain agents and LangGraph agents?+

LangChain AgentExecutor: Simple, linear loop. Good for basic ReAct. Black-box execution — hard to customize the flow. No persistent state. No parallelism. Limited control.

LangGraph agents: Explicit state machine. Every node and edge is defined by you. Full control over the flow. Native support for: branching (if/else routing), loops (ReAct cycle), parallel branches, human-in-the-loop (interrupt_before), persistent state across sessions (checkpointers), sub-graphs (nested agents).

When to use which: Simple agent that calls a few tools → AgentExecutor (less code). Complex workflows with branching logic, human approval, multi-agent collaboration, or session persistence → LangGraph. For production, LangGraph is almost always better — more debuggable, more controllable, more scalable.

MCP & Composio (Q23–Q26)

Q23What is MCP and why has it been called a "game-changer"?+

MCP (Model Context Protocol) is an open standard by Anthropic for connecting AI models to tools and data sources using a uniform JSON-RPC 2.0 protocol.

Problem before MCP: Every AI framework had its own tool format. LangChain tools couldn't be used in the OpenAI Assistants API. Cursor IDE had its own integration format. Every team wrote custom integrations that were incompatible.

Why game-changing: (1) Interoperability — build one MCP server, use it with Claude, GPT-4, LangGraph, Cursor, VS Code Copilot, any host. Write once, run anywhere. (2) Ecosystem — community builds MCP servers for common services, you install and use, no custom code. (3) Industry adoption — OpenAI, Anthropic, Microsoft GitHub Copilot, Cursor, Replit all adopted MCP in 2025. Becoming the USB-C standard for AI integrations. (4) Three concepts — Tools (functions AI calls), Resources (data AI reads), Prompts (reusable templates).

Q24When would you build a custom MCP server vs using Composio?+

Use Composio when: You need to integrate with popular services (Gmail, Slack, GitHub, Notion, Jira, Salesforce). Composio has 250+ pre-built servers. No custom auth, rate limiting, or API wrapper code needed. Minutes vs weeks of setup. Perfect for 80% of use cases.

Build custom MCP server when: You need to integrate with your internal company API/database. You need custom business logic beyond what APIs expose (e.g., complex database queries combining multiple tables). You're exposing company data as resources (dashboards, reports). Privacy concerns — your internal data should not go through Composio's servers. Your integration isn't in Composio's catalog.

Best practice: Use Composio for external services, build custom MCP for internal systems. Combine both in MultiServerMCPClient.

Q27When do you use Bedrock vs SageMaker? Give a real-world scenario for each.+

Bedrock scenario: An e-commerce company wants to add an AI assistant that answers customer questions from their product documentation and support articles. They have no ML team. They need AWS compliance (HIPAA/SOC2). Deploy in 2 weeks. → Use Bedrock: call Claude API, use Bedrock Knowledge Bases for RAG, done.

SageMaker scenario: A medical diagnostics company has 500,000 labeled X-ray reports. Q28What is Bedrock Guardrails? When is it required?+

Bedrock Guardrails is an AWS-managed content safety and policy enforcement layer that runs before/after every LLM call. It operates independently of the model itself.

Features: (1) Content filtering — blocks hate speech, violence, sexual content at 5 configurable sensitivity levels. System Design (Q33–Q40)

Q33Design a multi-tenant AI SaaS that serves 1000 companies, each with their own private documents.+

Core challenge: Perfect data isolation — Company A must NEVER see Company B's documents.

Option A — Namespace isolation (recommended for most): One Pinecone/Qdrant index with namespaces per tenant. All upserts/queries use namespace: namespace=f"tenant_{company_id}". Simple, cheap, fast to implement. Risk: a code bug could cross-contaminate. Mitigate with defensive coding and tests.

Option B — Metadata filtering: Store tenant_id in every chunk's metadata. Always filter queries: where={"tenant_id": company_id}. Works well with Qdrant payload filtering. Same index, strict filtering.

Option C — Separate indexes (true isolation): One Pinecone/Qdrant index per company. Perfect isolation, zero blast radius. Very expensive at scale ($70+/month per company).

Auth: JWT tokens with tenant_id claim. Middleware extracts from token, never from request body. Every DB query, every retrieval call includes tenant_id filter as mandatory parameter.

Start with Option B (metadata), move to Option A (namespaces) at scale, offer Option C to enterprise tier customers.

Q34How would you reduce LLM API costs by 80% for a high-traffic application?+

Strategy 1 — Semantic Caching (saves 30-50%): Cache LLM responses. When a new query arrives, check if a semantically similar query was answered before (cosine similarity > 0.95). Return cached answer. Many users ask the same or similar questions. GPTCache or custom Redis + embedding implementation. Near-zero latency for cache hits.

Strategy 2 — Model Routing (saves 30-60%): Classify query complexity before calling the LLM. Simple factual queries → GPT-4o-mini ($0.15/1M tokens). Complex multi-step reasoning → GPT-4o ($5/1M). That's a 33x cost difference. Route 70% of queries to the cheap model.

Strategy 3 — Prompt Compression (saves 20-30%): Use LLMLingua to compress system prompts and retrieved context by 50% with minimal quality degradation. Fewer input tokens = lower cost.

Strategy 4 — Better Retrieval (saves 20-40%): Retrieve 3 high-quality chunks instead of 10 mediocre ones. Reranking helps. Fewer context tokens, same or better answer quality.

Strategy 5 — Batch Processing (saves 10-20%): Use OpenAI Batch API for non-time-sensitive tasks. 50% cost discount for async batch jobs (results returned within 24h).

Q35How do you handle LLM rate limits and API failures in production?+

LLM APIs fail more often than regular REST APIs. Rate limits, timeouts, and transient errors are common at scale.

1. Exponential backoff with jitter: On RateLimitError or 5xx, retry with wait = 2^attempt + random(0,1). This spreads out retries to avoid thundering herd.

2. Fallback providers: Primary fails → secondary. Bedrock Claude → OpenAI GPT-4o → GPT-4o-mini (cheaper fallback). Implement with try/except chain.

3. Circuit breaker: If primary provider fails 5 times in 60 seconds, "open the circuit" — route all traffic to fallback for the next 2 minutes. Prevents hammering a failing service.

4. Timeout handling: Set hard timeout (30 seconds). If timeout, return a graceful message: "Response is taking longer than expected. Please try again." Don't leave the user hanging.

5. Queue for heavy loads: Put document indexing and batch LLM jobs in a queue (Celery + Redis). Don't block the API thread on slow operations.

from langchain_openai import OpenAIEmbeddings
import numpy as np
from typing import Optional, List, Tuple

class SemanticCache:
    def __init__(self, similarity_threshold: float = 0.95):
        self.embeddings = OpenAIEmbeddings()
        self.threshold  = similarity_threshold
        self.cache: List[Tuple[List[float], str, str]] = []

    def get(self, question: str) -> Optional[str]:
        if not self.cache: return None
        q_emb = self.embeddings.embed_query(question)
        for stored_emb, _, answer in self.cache:
            sim = np.dot(q_emb, stored_emb) / (
                np.linalg.norm(q_emb) * np.linalg.norm(stored_emb))
            if sim >= self.threshold:
                print(f"Cache hit! similarity={sim:.3f}")
                return answer
        return None

    def set(self, question: str, answer: str):
        emb = self.embeddings.embed_query(question)
        self.cache.append((emb, question, answer))

cache = SemanticCache(similarity_threshold=0.95)

    hit = cache.get(question)
    if hit: return hit  # free!
    answer = llm.invoke(question).content  # paid API call
    cache.set(question, answer)
    return answer

Q37CODE: Build a simple document Q&A pipeline from scratch (no LangChain)+

from openai import OpenAI
import numpy as np

client = OpenAI()

def embed(text: str) -> List[float]:
    res = client.embeddings.create(model="text-embedding-3-small", input=text)
    return res.data[0].embedding

def similarity(a, b) -> float:
    va, vb = np.array(a), np.array(b)
    return float(np.dot(va, vb) / (np.linalg.norm(va) * np.linalg.norm(vb)))

def chunk_text(text: str, size: int = 500, overlap: int = 100) -> List[str]:
    chunks = []
    for i in range(0, len(text), size - overlap):
        chunks.append(text[i : i + size])
    return chunks

# Index documents
documents = ["Employees get 20 vacation days per year.",
             "Remote work is allowed 3 days a week.",
             "Health insurance covers dental and vision."]

index = [(doc, embed(doc)) for doc in documents]

def answer(question: str, k: int = 2) -> str:
    q_emb = embed(question)
    ranked = sorted(index, key=lambda x: similarity(q_emb, x[1]), reverse=True)
    context = "\n".join(doc for doc, _ in ranked[:k])
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {"role": "system", "content": f"Answer from context only:\n{context}"},
            {"role": "user",   "content": question}
        ]
    )
    return response.choices[0].message.content

print(answer("How many vacation days do I get?"))
# "You get 20 vacation days per year."

Do This / Avoid This

✅ Do This
  • Start with the simplest working solution, then add complexity
  • Know when NOT to use LangChain (simple single API calls)
  • Talk about cost optimization, monitoring, and observability in every design
  • Mention RAGAS by name when asked about RAG evaluation
  • Have deployed projects — not just Jupyter notebooks
  • Read LangChain and LangGraph docs weekly — they change fast
  • Practice explaining technical decisions to a non-technical audience
❌ Avoid This
  • Using LangChain for simple one-LLM calls (adds overhead for no benefit)
  • Not handling LLM errors, rate limits, and timeouts
  • Storing API keys in code, git repos, or environment (use Secrets Manager)
  • Using in-memory chat history or in-memory vector store in production
  • Not measuring quality (blind optimism about LLM output quality)
  • Over-engineering with LangGraph when a simple chain would do
  • Skipping Docker and going straight to cloud — local Docker first always
